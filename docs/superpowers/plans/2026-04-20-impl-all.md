### Task 3.3 · ActivityStatusService（纯函数 + 单测）

**关键代码**：

```ts
// transitions.ts
export type Transition = { from: ActivityStatus; event: TransitionEvent; guard?: (ctx: TransitionCtx) => boolean; to: ActivityStatus };

export function computeNextStatus(current: ActivityStatus, event: TransitionEvent, ctx: TransitionCtx): ActivityStatus | null {
  const table: Transition[] = [
    { from: 'WAITING_FOR_MEMBERS', event: 'JOIN', guard: ctx => ctx.participantCount + 1 === ctx.maxParticipants, to: 'FORMED' },
    { from: 'WAITING_FOR_MEMBERS', event: 'JOIN', guard: ctx => ctx.participantCount + 1 < ctx.maxParticipants, to: 'WAITING_FOR_MEMBERS' },
    { from: 'WAITING_FOR_MEMBERS', event: 'CRON_TICK', guard: ctx => ctx.now > ctx.startTime, to: 'CANCELLED' },
    { from: 'FORMED', event: 'LEAVE', to: 'WAITING_FOR_MEMBERS' },
    { from: 'FORMED', event: 'CRON_TICK', guard: ctx => ctx.now >= new Date(ctx.startTime.getTime() - 60 * 60 * 1000), to: 'STARTING_SOON' },
    { from: 'STARTING_SOON', event: 'CRON_TICK', guard: ctx => ctx.now >= ctx.startTime, to: 'IN_PROGRESS' },
    { from: 'IN_PROGRESS', event: 'CRON_TICK', guard: ctx => ctx.now >= new Date(ctx.startTime.getTime() + 4 * 60 * 60 * 1000), to: 'COMPLETED' },
    // CANCEL 从任意非 COMPLETED → CANCELLED
  ];
  const m = table.find(t => t.from === current && t.event === event && (!t.guard || t.guard(ctx)));
  return m?.to ?? null;
}
```
